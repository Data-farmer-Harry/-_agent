# MatterPilot portable Python requirements
#
# Recommended runtime:
#   Python 3.12
#   pip from a Conda or virtualenv environment
#
# Portable install example:
#   conda create -n lammps_agent python=3.12 pip
#   conda activate lammps_agent
#   python -m pip install -r requirements-portable.txt
#
# Frontend dependencies are intentionally not listed here because they are
# managed by frontend/package-lock.json. For the frontend, run:
#   cd frontend && npm ci
#
# Host tools that are not Python packages:
#   LAMMPS executable, configured via LAMMPS_CMD or the runtime settings UI
#   ffmpeg, used for browser-friendly trajectory media conversion
#   EAM potential files, configured via POTENTIALS_DIR
#
# OVITO is included to preserve current visualization capability. If a target
# platform cannot install OVITO or you do not need GIF/MP4 trajectory rendering,
# comment out ovito==3.15.1 and use an external ovitos executable instead.
